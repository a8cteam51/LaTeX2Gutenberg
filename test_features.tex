\section{Display Math Line Breaking}

% Long display math formulas (over 100 characters) should be
% automatically split into two consecutive math blocks at a
% relation operator.  The second block starts with the operator
% (e.g.\ \leq, \geq), matching the style on Terry Tao's blog.

Short formula (should stay on one line):
$$\prod_{i=1}^k \left(1 - \frac{1}{q_i}\right).$$

Long formula with \verb|\leq| (should split into two blocks):
$$|\{ n \in \Z/Q\Z: n \neq a_i \hbox{ mod } q_i \hbox{ for all } i=1,\ldots,k \}| \leq |\{ n \in \Z/Q\Z: n \neq 0 \hbox{ mod } q_i \hbox{ for all } i=1,\ldots,k \}|.$$

Long formula with \verb|\geq| (should split into two blocks):
$$ |\bigcup_{j=1}^k (a_{j,1} + H_{j,1}) \times (a_{j,2} + H_{j,2})| \geq |\bigcup_{j=1}^k (a_{j,1} + H_{j,1}) \times H_{j,2}|$$

\section{Diagram Environment}

% The {diagram} environment (Paul Taylor's diagrams package)
% should convert to a \begin{array} math block.  Arrow commands
% are mapped to standard LaTeX:
%   \rTo -> \rightarrow    \lTo -> \leftarrow
%   \dTo -> \downarrow     \uTo -> \uparrow
%   \rdTo -> \searrow      \ruTo -> \nearrow
%   \ldTo -> \swarrow      \luTo -> \nwarrow
% Superscript/subscript labels (e.g.\ \rTo^{f}) are preserved.

Bare diagram environment:

\begin{diagram}
A & \rTo^{f} & B \\
\dTo^{g} &     & \dTo_{h} \\
C & \rTo_{k} & D
\end{diagram}

Diagram wrapped in display math delimiters (should not produce
a stray \verb|\]| in the output):

\[
\begin{diagram}
X & \rTo^{\alpha} & Y \\
\dTo^{\beta} &     & \dTo_{\gamma} \\
Z & \rTo_{\delta} & W
\end{diagram}
\]

\section{Inline Math MathML Fixes}

% \mathbb letters in inline math should use literal Unicode double-struck
% characters (e.g.\ U+2124 for Z) rather than mathvariant="double-struck",
% which is unreliable across browsers.

Inline: $n \in \mathbb{Z}$, $x \in \mathbb{R}$, $z \in \mathbb{C}$,
$p \in \mathbb{N}$, $q \in \mathbb{Q}$.

% \hbox{ mod } should not produce 1em gaps around "mod".  The spaces
% inside \hbox are converted by latex2mathml to &nbsp;; we strip them
% so the spacing matches surrounding math operators (~0.33em).

Spacing: $a_i \hbox{ mod } q_i$ should have normal word spacing around ``mod''.

\section{Tikzcd Commutative Diagrams}

% The {tikzcd} environment should be converted to a
% \begin{array} math block by expanding the node grid
% and placing arrow symbols between nodes.

\begin{tikzcd}

C \arrow[r, "k"'] & D
\end{tikzcd}

Diagonal arrow example:

\begin{tikzcd}
A \arrow[r, "f"] \arrow[dr, "\phi"'] & B \arrow[d, "g"] \\
 & C
\end{tikzcd}

\[
\begin{tikzcd}
X \arrow[r, "\alpha"] \arrow[d, "\beta"'] & Y \arrow[d, "\gamma"] \\
Z \arrow[r, "\delta"'] & W

\]
